# Dependencias de KeyServ (Fase 3 — antes no existía este archivo).
#
# Núcleo del backend Django — necesario para levantar el sitio.
# Django==4.2.1 (la versión original, pinneada por la tesis) tiene un bug de
# incompatibilidad real con Python 3.14 en su test client (django/template/context.py
# __copy__ revienta con "AttributeError: 'super' object has no attribute 'dicts'"
# en CADA request hecho por `manage.py test` — no es un bug de KeyServ, es de
# Django). Se probó primero con el último parche 4.2.30 (misma rama LTS) y
# seguía roto. Se subió a Django 5.2 LTS (Fase 4), que sí soporta Python 3.14
# limpio — los 21 tests automatizados corren sin errores con esta versión.
# django-environ/psycopg2-binary/Pillow sin pin exacto a propósito: se probó
# instalar en Python 3.14 (venv_tesis) y las versiones ancla originales no
# tenían wheel precompilado para esa versión de Python tan nueva; con
# mínimos quedan libres para resolver la última compatible.
Django==5.2.16
django-environ>=0.11
psycopg2-binary>=2.9.9
Pillow>=10.3

# Geolocalización aproximada por IP para IntentoAccesoSospechoso (ver
# codigo/backend/django/KeyServApp/geolocalizacion.py) — lee una base DB-IP
# City Lite local (`manage.py descargar_geoip`, gratuita, sin cuenta).
geoip2>=4.8

# Biometría de huella dactilar (codigo/biometria/huella/IMAGEN_HUELLA.py) —
# solo necesita Pillow, ya declarado arriba.

# Reconocimiento facial (codigo/biometria/reconocimiento_facial/) — pesado
# (dlib se compila nativamente, necesita el workload "Desktop development
# with C++" de Visual Studio Build Tools instalado en la máquina), instalar
# aparte si se va a probar esa funcionalidad puntual. No es necesario para
# levantar el resto del sitio.
# opencv-python==4.9.0.80 (la versión que pedía la tesis) está compilado
# contra la ABI de NumPy 1.x y revienta con "AttributeError: _ARRAY_API not
# found" bajo NumPy 2.x — y NumPy 1.x no tiene wheel para Python 3.14, así
# que no se puede bajar NumPy en vez de subir opencv. Se sube opencv-python
# a la primera versión con wheel para cp314 (>=4.10 ya soporta NumPy 2.x).
opencv-python>=4.10
face_recognition==1.3.0
# face_recognition_models (dependencia de face_recognition) todavía usa
# `pkg_resources` para ubicar sus archivos .dat, y setuptools>=81 lo quitó
# por completo (antes solo lo marcaba deprecado) — sin este pin, importar
# face_recognition_models revienta con "ModuleNotFoundError: No module
# named 'pkg_resources'" pese a tener setuptools instalado.
setuptools<81

# Generación de PDF (codigo/biometria/huella/GUARDAR_DOCUMENTO.py)
fpdf2==2.7.9

# Pagos (codigo/backend/django/KeyServApp/pagos.py) — Webpay Plus (tarjetas)
# + Khipu (transferencia bancaria). Webpay se prueba de punta a punta con
# las credenciales públicas de integración del propio SDK, sin cuenta real;
# Khipu usa `requests` (ya declarado arriba, sin SDK de terceros — ver
# pagos.py sobre por qué) y sí necesita cuenta real para probarse.
transbank-sdk>=6.1.0

# Escaneo antivirus de archivos subidos (codigo/backend/django/KeyServApp/antivirus.py)
